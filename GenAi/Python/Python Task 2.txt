2.Print odd numbers from 1 to 10
		i = 1
		while i <=10:
		  if i%2 !=0:
			print(i)
		  i = i + 1	
		  
3.Print numbers greater than 5 from 1 to 10
		i = 1

4.Print numbers from 1 to 20, excluding 5
		i = 1

		  i = i+1
		  
5.Count numbers divisible by both 3 and 5 between 1 and 100
		i = 1
		count = 0
		while i <= 100:
		  if i%3==0 and i%5 ==0:
			count = count+1
		  i = i +1
		print(count)

7.Find the sum of numbers divisible by 5 between 1 and 50

		i = 1
		sum = 0
		while i<=50:

		print(sum)
